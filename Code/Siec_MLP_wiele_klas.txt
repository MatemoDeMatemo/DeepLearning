import math

#### Wagi po treningu (wklejasz tutaj wyniki z kodu treningowego) ####
W1 = [
    [0.5, -0.3],
    [0.8,  0.2]
]
B1 = [[0.1], [-0.2]]

W2 = [
    [0.7, -0.5],
    [0.2,  0.9],
    [-0.3, 0.4]
]
B2 = [[0.05], [-0.1], [0.2]]

#### Dane wejściowe ####
X = [2, 5]

#### Funkcje  ####
def matmul(W, X):
    return [sum(W[i][j]*X[j] for j in range(len(X))) for i in range(len(W))]

def add_bias(Z, B):
    return [Z[i] + B[i][0] for i in range(len(Z))]

def relu(Z):
    return [z if z > 0 else 0 for z in Z]    # ReLU

def softmax(Z):
    max_z = max(Z)
    exp_vals = [math.exp(z - max_z) for z in Z]
    s = sum(exp_vals)
    return [e/s for e in exp_vals]

#### Forward pass — krok po kroku ####
H_lin  = matmul(W1, X)
H_bias = add_bias(H_lin, B1)
H      = relu(H_bias)

Z_lin  = matmul(W2, H)
Z      = add_bias(Z_lin, B2)
Y      = softmax(Z)

#### Output ####
print("WEJŚCIE X:")
print(X)

print("\nWARSTWA UKRYTA (przed aktywacją):")
print(H_bias)

print("\nWARSTWA UKRYTA (po ReLU):")
print(H)

print("\nLOGITY Z:")
print(Z)

print("\nSOFTMAX OUTPUT:")
print(Y)

print("\nDECYZJA: klasa", Y.index(max(Y)) + 1)
