#### 1] Dane wejściowe (wektor = macierz kolumnowa) ####

X = [
    [2],
    [5]
]


#### 2] Warstwa ukryta (W1 * X + b1) ####

W1 = [
    [0.5, -0.3],
    [0.8,  0.2]
]

B1 = [
    [0.1],
    [-0.2]
]


#### Funkcja mnożenia macierzy przez wektor ####
def matmul(W, X):
    wynik = []
    for i in range(len(W)):
        suma = 0
        for j in range(len(X)):
            suma += W[i][j] * X[j][0]
        wynik.append([suma])
    return wynik


#### Dodawanie biasu ####
def add_bias(Z, B):
    wynik = []
    for i in range(len(Z)):
        wynik.append([Z[i][0] + B[i][0]])
    return wynik


#### Warstwa ukryta ####

H_lin = matmul(W1, X)      # W1 * X
H = add_bias(H_lin, B1)    # + B1


#### 3] Warstwa wyjściowa (W2 * H + b2) ####

W2 = [
    [0.7, -0.5],
    [0.2,  0.9],
    [-0.3, 0.4]
]

B2 = [
    [0.05],
    [-0.1],
    [0.2]
]


Z_lin = matmul(W2, H)
Z = add_bias(Z_lin, B2)


#### 4] Softmax ####

import math

def softmax(Z):
    max_z = max([z[0] for z in Z])

    exp_vals = []
    for i in range(len(Z)):
        exp_vals.append(math.exp(Z[i][0] - max_z))
    
    suma = sum(exp_vals)

    wynik = []
    for i in range(len(exp_vals)):
        wynik.append(exp_vals[i] / suma)
    
    return wynik


Y = softmax(Z)

#### 5] Output ####

print("WEJŚCIE X:")
print(X)

print("\nWARSTWA UKRYTA H = W1·X + B1:")
print(H)

print("\nLOGITY Z = W2·H + B2:")
print(Z)

print("\nSOFTMAX OUTPUT:")
print(Y)
